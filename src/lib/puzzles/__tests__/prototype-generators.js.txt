function seedFrom(str){ let h = 2166136261>>>0; for(const ch of String(str)){ h ^= ch.charCodeAt(0); h = Math.imul(h, 16777619)>>>0; } return h; }
function rng(seed){ let a = seed>>>0; return ()=>{ a = (a + 0x6D2B79F5)>>>0; let t = a; t = Math.imul(t ^ t>>>15, t|1); t ^= t + Math.imul(t ^ t>>>7, t|61); return ((t ^ t>>>14)>>>0) / 4294967296; }; }
function shuffled(arr, r){ const a = arr.slice(); for(let i=a.length-1;i>0;i--){ const j = Math.floor(r()*(i+1)); [a[i],a[j]] = [a[j],a[i]]; } return a; }

const CW_LEVELS = {
  1:{n:6,  tiers:[1],         max:6,  size:9},
  2:{n:7,  tiers:[1,1,1,2],   max:7,  size:9},
  3:{n:9,  tiers:[1,2,2],     max:8,  size:11},
  4:{n:10, tiers:[2,2,3],     max:9,  size:11},
  5:{n:12, tiers:[2,3,3,3],   max:10, size:11}
};
function cwLayout(cands, cfg, r){
  const cells = new Map(), K = (a,b)=>a+","+b, words = [];
  let minR=0, maxR=0, minC=0, maxC=0, inters=0;
  const fits = (w, sr, sc, dir)=>{
    const dr = dir==="d"?1:0, dc = dir==="a"?1:0, L = w.length;
    if(cells.has(K(sr-dr, sc-dc)) || cells.has(K(sr+dr*L, sc+dc*L))) return -1;
    let hit = 0;
    for(let i=0;i<L;i++){
      const rr = sr+dr*i, cc = sc+dc*i, cell = cells.get(K(rr,cc));
      if(cell){ if(cell.ch!==w[i] || cell[dir]) return -1; hit++; }
      else if(dir==="a" ? (cells.has(K(rr-1,cc))||cells.has(K(rr+1,cc))) : (cells.has(K(rr,cc-1))||cells.has(K(rr,cc+1)))) return -1;
    }
    const a0 = Math.min(minR,sr), a1 = Math.max(maxR, sr+dr*(L-1)), b0 = Math.min(minC,sc), b1 = Math.max(maxC, sc+dc*(L-1));
    if(a1-a0+1 > cfg.size || b1-b0+1 > cfg.size) return -1;
    if(!hit || hit===L) return -1;
    return hit*10 - ((a1-a0+1)*(b1-b0+1) - (maxR-minR+1)*(maxC-minC+1))*0.3 + r()*3;
  };
  const put = (e, sr, sc, dir)=>{
    const dr = dir==="d"?1:0, dc = dir==="a"?1:0;
    for(let i=0;i<e.w.length;i++){
      const k = K(sr+dr*i, sc+dc*i), cell = cells.get(k) || {ch:e.w[i]};
      if(cell.a||cell.d) inters++;
      cell[dir] = true; cells.set(k, cell);
    }
    minR = Math.min(minR,sr); maxR = Math.max(maxR, sr+dr*(e.w.length-1)); minC = Math.min(minC,sc); maxC = Math.max(maxC, sc+dc*(e.w.length-1));
    words.push({w:e.w, c:e.c, r:sr, col:sc, dir});
  };
  put(cands[0], 0, 0, r()<.5?"a":"d");
  for(const e of cands.slice(1)){
    if(words.length >= cfg.n) break;
    if(words.some(x=>x.w===e.w)) continue;
    let best = null;
    for(const pw of words){
      const nd = pw.dir==="a"?"d":"a";
      for(let i=0;i<pw.w.length;i++) for(let j=0;j<e.w.length;j++){
        if(pw.w[i]!==e.w[j]) continue;
        const cr = pw.r + (pw.dir==="d"?i:0), cc = pw.col + (pw.dir==="a"?i:0);
        const sr = cr - (nd==="d"?j:0), sc = cc - (nd==="a"?j:0);
        const sc0 = fits(e.w, sr, sc, nd);
        if(sc0>=0 && (!best || sc0>best.s)) best = {s:sc0, sr, sc, nd};
      }
    }
    if(best) put(e, best.sr, best.sc, best.nd);
  }
  return {words, inters, minR, minC, rows:maxR-minR+1, cols:maxC-minC+1, score: words.length*100 + inters*12 - (maxR-minR+1)*(maxC-minC+1)*0.5};
}
function makeCrossword(level, seedStr){
  const cfg = CW_LEVELS[level] || CW_LEVELS[1], r = rng(seedFrom(seedStr));
  const byTier = {}; [1,2,3].forEach(t=>byTier[t] = shuffled(CW_BANK.filter(e=>e.t===t && e.w.length>=3 && e.w.length<=cfg.max), r));
  let best = null;
  for(let attempt=0; attempt<36; attempt++){
    const used = new Set(), cands = [];
    while(cands.length < cfg.n*5){
      const t = cfg.tiers[Math.floor(r()*cfg.tiers.length)], list = byTier[t];
      const e = list[Math.floor(r()*list.length)];
      if(!used.has(e.w)){ used.add(e.w); cands.push(e); }
    }
    const longFirst = cands.slice().sort((a,b)=>b.w.length-a.w.length);
    const start = longFirst.splice(Math.floor(r()*3),1)[0];
    const res = cwLayout([start, ...shuffled(longFirst, r)], cfg, r);
    if(!best || res.score > best.score) best = res;
    if(best.words.length >= cfg.n && attempt >= 12) break;
  }
  const words = best.words.map(x=>({w:x.w, c:x.c, r:x.r-best.minR, col:x.col-best.minC, dir:x.dir}));
  const starts = [...new Set(words.map(x=>x.r*100+x.col))].sort((a,b)=>a-b);
  words.forEach(x=>x.num = starts.indexOf(x.r*100+x.col)+1);
  words.sort((a,b)=> (a.dir===b.dir ? a.num-b.num : a.dir==="a"?-1:1));
  const sol = {}, at = {};
  words.forEach((x,wi)=>{ for(let i=0;i<x.w.length;i++){ const k = (x.r+(x.dir==="d"?i:0))+","+(x.col+(x.dir==="a"?i:0)); sol[k] = x.w[i]; (at[k] = at[k]||{})[x.dir] = wi; } });
  const nums = {}; words.forEach(x=>nums[x.r+","+x.col] = x.num);
  return {level, rows:best.rows, cols:best.cols, words, sol, at, nums};
}

/* sudoku: seeded full grid, then remove clues while the answer stays unique */
function suSolve(g, limit, r){
  const rows = new Array(9).fill(0), cols = new Array(9).fill(0), boxes = new Array(9).fill(0);
  const bx = i => ((i/27)|0)*3 + (((i%9)/3)|0);
  for(let i=0;i<81;i++) if(g[i]){ const b = 1<<g[i]; rows[(i/9)|0] |= b; cols[i%9] |= b; boxes[bx(i)] |= b; }
  let count = 0;
  const rec = ()=>{
    let best = -1, bm = 0, bn = 10;
    for(let i=0;i<81;i++) if(!g[i]){
      const m = ~(rows[(i/9)|0] | cols[i%9] | boxes[bx(i)]) & 0x3FE;
      let n = 0; for(let x=m; x; x &= x-1) n++;
      if(n < bn){ bn = n; best = i; bm = m; if(!n) return false; }
    }
    if(best < 0){ count++; return count >= limit; }
    let ds = []; for(let d=1; d<=9; d++) if(bm & (1<<d)) ds.push(d);
    if(r) ds = shuffled(ds, r);
    const ri = (best/9)|0, ci = best%9, bi = bx(best);
    for(const d of ds){
      const b = 1<<d; g[best] = d; rows[ri] |= b; cols[ci] |= b; boxes[bi] |= b;
      if(rec()) return true;
      g[best] = 0; rows[ri] &= ~b; cols[ci] &= ~b; boxes[bi] &= ~b;
    }
    return false;
  };
  rec();
  return count;
}
const SU_CLUES = {1:44, 2:37, 3:32, 4:28, 5:25};
function makeSudoku(level, seedStr){
  const r = rng(seedFrom(seedStr)), g = new Array(81).fill(0);
  suSolve(g, 1, r);
  const sol = g.slice(), target = SU_CLUES[level] || 44;
  let clues = 81;
  for(const i of shuffled([...Array(81).keys()], r)){
    if(clues <= target) break;
    const v = g[i]; g[i] = 0;
    if(suSolve(g.slice(), 2) !== 1) g[i] = v; else clues--;
  }
  return {level, puzzle:g.join(""), solution:sol.join(""), clues};
}
